def main():
    print("--- Эй чел введи циффрыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыыы ---")
    try:
        num1 = float(input("Введите первое число: "))
        num2 = float(input("Введите второе число: "))
        
        result = (num1 * num2)*num1
        
        # Используем f-строку для вывода (доступно в современном Python)
        print(f"Результат: ({num1} + {num2})*{num1} = {result}")
    except ValueError:
        print("Ошибка: пожалуйста, вводите только числа.")

if __name__ == "__main__":
    main()
