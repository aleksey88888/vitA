import matplotlib.pyplot as plt
import matplotlib.patches as patches

# 1. Задаем цвета в разных форматах
colors = {
    "Red (Name)": "red",
    "Green (HEX)": "#2ECC71",
    "Blue (RGB)": (0.2, 0.4, 0.8),  # В matplotlib RGB от 0 до 1
    "Custom Orange": "#E67E22",
    "Purple": "#9B59B6"
}

# 2. Создаем график для визуализации
fig, ax = plt.subplots(figsize=(8, 4))
ax.set_xlim(0, len(colors))
ax.set_ylim(0, 1)

# 3. Рисуем прямоугольники с этими цветами
for i, (name, color) in enumerate(colors.items()):
    # Рисуем цветной блок
    rect = patches.Rectangle((i, 0), 1, 1, color=color)
    ax.add_patch(rect)
    # Добавляем подпись
    ax.text(i + 0.5, 0.5, name, color='white', weight='bold',
            ha='center', va='center', fontsize=10, bbox=dict(facecolor='black', alpha=0.3))

# Убираем оси
ax.set_xticks([])
ax.set_yticks([])
plt.title("Визуализация цветов в Python", fontsize=14)
plt.show()
